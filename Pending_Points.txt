# TextGuru Send SMS API — Python 3 (requests)
import requests

url = 'https://www.textguru.in/api/v22.0/'

payload = {
    'username'  : 'smvs.org',
    'password'  : 'your_password',
    'source'    : 'TGURUS',
    'dmobile'   : '918284047608,918284047606',
    'dlttempid' : '1707160248729731',
    'message'   : 'Your OTP is 1234. Valid 10 mins.',
}

try:
    r = requests.post(url, data=payload, timeout=30)
    r.raise_for_status()
    print("Response:", r.text)
except requests.exceptions.RequestException as e:
    print("Error:", e)

API Response Codes
Common Response Codes & Meanings
Code	Status	Meaning
1701	Success	SMS submitted successfully to the operator
1702	Failed	Invalid username or password
1703	Failed	Invalid or mismatched DLT Template ID
1704	Failed	Invalid or malformed mobile number
1705	Failed	Insufficient credits — please recharge your account
1706	Failed	Sender ID / Header not registered or blocked by operator
1707	Failed	Account blocked or suspended — contact support
